cmake_minimum_required(VERSION 3.25)
project(QIS_X VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "QIS-X requires Windows and Direct3D 11.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/shaders.hlsl)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/shaders.hlsl" QIS_SHADER_SOURCE)
configure_file(cmake/ShaderSource.h.in generated/ShaderSource.h @ONLY)
set(QIS_COMPILE_DEFINITIONS UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
# Visual Studio generators do not emit compile_commands.json. Supply a fixed
# compilation database so clangd sees the same language, includes and defines.
set(QIS_EDITOR_FLAGS "--driver-mode=cl\n/std:c++17\n/EHsc\n/I${CMAKE_CURRENT_SOURCE_DIR}/include\n/I${CMAKE_CURRENT_BINARY_DIR}/generated\n")
foreach(definition IN LISTS QIS_COMPILE_DEFINITIONS)
    string(APPEND QIS_EDITOR_FLAGS "/D${definition}\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/compile_flags.txt" CONTENT "${QIS_EDITOR_FLAGS}")
add_library(qis_core STATIC src/Application.cpp src/Renderer.cpp src/FrameSync.cpp src/Texture.cpp src/WICTextureLoader.cpp)
target_include_directories(qis_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(qis_core PUBLIC ${QIS_COMPILE_DEFINITIONS})
target_link_libraries(qis_core PUBLIC d3d11 dxgi d3dcompiler d2d1 dwrite windowscodecs ole32 uuid comdlg32 shell32 dwmapi)
if(MSVC)
    target_compile_options(qis_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
else()
    target_compile_options(qis_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(QIS-X WIN32 QIS_X-V.1.cpp resources/app.rc)
target_link_libraries(QIS-X PRIVATE qis_core)
if(MSVC)
    target_link_options(QIS-X PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
    target_link_options(QIS-X PRIVATE -municode -static-libgcc -static-libstdc++)
endif()
function(qis_copy_sample target)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/assets"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/test_image.png" "$<TARGET_FILE_DIR:${target}>/assets/demo.png"
        VERBATIM)
endfunction()
qis_copy_sample(QIS-X)
include(CTest)
if(BUILD_TESTING)
    add_executable(qis-tests tests/PrototypeTests.cpp)
    target_link_libraries(qis-tests PRIVATE qis_core)
    qis_copy_sample(qis-tests)
    add_test(NAME prototype-pipeline COMMAND qis-tests "${CMAKE_CURRENT_BINARY_DIR}/test-artifacts")
endif()
install(TARGETS QIS-X RUNTIME DESTINATION .)
install(FILES test_image.png DESTINATION assets RENAME demo.png)
install(FILES README.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
